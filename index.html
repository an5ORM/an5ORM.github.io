Quick Start

Up and running in minutes

# Install
npm install @an5/orm @an5/adapters

# Define schema (an5Schema/User.an5)
model User {
  id        NVARCHAR(1000)   @id @default(uuid())
  email     NVARCHAR(255)    @unique
  name      NVARCHAR(255)?
  createdAt DATETIME2        @default(now())

  @@map("users")
}

# Generate the typed clients (from an5Orm/)
npm run generate

# Use
import { createAn5Adapter } from '@an5/adapters';

const db = createAn5Adapter({ connectionString: process.env.DATABASE_URL });
const users = await db.user.findMany({ take: 10 });
# Install
pip install an5-adapters

# Use the adapter (or the generated An5Client)
import os
from an5_adapters import create_an5_adapter

db = create_an5_adapter(os.environ["DATABASE_URL"])

users = db.table("User").find_many(where={"isActive": True})
using An5Orm;

var db = new An5DbContext(
  "Server=.;Database=mydb;Trusted_Connection=true"
);

// Create
var user = new User { Email = "john@example.com", Name = "John" };
db.User.Create(user);

// Read (synchronous TableClient)
var users = db.User.FindMany(
  "Email = @email",
  new Dictionary<string, object> { ["email"] = "john@example.com" }
);
// Generated an5client with type-safe args
import (
  "context"
  "database/sql"
  an5 "an5client"
  _ "github.com/denisenkom/go-mssqldb"
)

sqlDB, _ := sql.Open("sqlserver", an5.GetDefaultConnectionString())
orm := an5.NewAn5DbContext(sqlDB)
ctx := context.Background()

// Find users matching an email pattern
users, err := orm.User.FindMany(ctx, &an5.UserFindManyArgs{
  Where: &an5.UserWhereInput{
    Email: &an5.StringFilter{Contains: an5.StringPtr("@example.com")},
  },
  Take: an5.IntPtr(10),
})
// Browser-compatible! No Node.js builtins
import { createAn5SheetsAdapter } from '@an5/adapters/browser';

const db = createAn5SheetsAdapter({
  spreadsheetId: 'YOUR_SHEET_ID',
  accessToken: 'YOUR_OAUTH_TOKEN'
  // or apiKey: 'YOUR_API_KEY'
});

const rows = await db.table('Products').findMany();
